cmake_minimum_required(VERSION 3.8)
project(gp7_grasp)
find_package(ament_cmake REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(gp7_grasp-system SHARED src/validated_grasp.cpp)
target_compile_features(gp7_grasp-system PUBLIC cxx_std_17)
target_link_libraries(gp7_grasp-system gz-sim8::gz-sim8 gz-plugin2::register)
install(TARGETS gp7_grasp-system LIBRARY DESTINATION lib)
ament_package()
